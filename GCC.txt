#include <stdio.h>

void bubbleSort(int arr[], int n)
{
    for (int i = 0; i < n - 1; i++)
    {
        for (int j = 0; j < n - i - 1; j++)
        {
            if (arr[j] > arr[j + 1])
            {
                int t = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = t;
            }
        }
    }
}

void insertionSort(int arr[], int n)
{
    for (int i = 1; i < n; i++)
    {
        int key = arr[i];
        int j = i - 1;

        while (j >= 0 && arr[j] > key)
        {
            arr[j + 1] = arr[j];
            j--;
        }

        arr[j + 1] = key;
    }
}

int partition(int arr[], int low, int high)
{
    int pivot = arr[high];
    int i = low - 1;

    for (int j = low; j < high; j++)
    {
        if (arr[j] <= pivot)
        {
            i++;

            int temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
        }
    }

    int temp = arr[i + 1];
    arr[i + 1] = arr[high];
    arr[high] = temp;

    return i + 1;
}

void quickSort(int arr[], int low, int high)
{
    if (low < high)
    {
        int pivotIndex = partition(arr, low, high);

        quickSort(arr, low, pivotIndex - 1);
        quickSort(arr, pivotIndex + 1, high);
    }
}

void sortMatrix()
{
    int i, j;
    struct Element temp;

    for (i = 0; i < count - 1; i++)
    {
        for (j = 0; j < count - i - 1; j++)
        {
            if (sparse[j].row > sparse[j + 1].row ||
                (sparse[j].row == sparse[j + 1].row &&
                 sparse[j].col > sparse[j + 1].col))
            {
                temp = sparse[j];
                sparse[j] = sparse[j + 1];
                sparse[j + 1] = temp;
            }
        }
    }

    printf("Sparse matrix sorted successfully.\n");
}

{
    int i;
    struct Element temp;

    for (i = 0; i < count; i++)
    {
        temp.row = sparse[i].row;
        sparse[i].row = sparse[i].col;
        sparse[i].col = temp.row;
    }

    temp.row = rows;
    rows = cols;
    cols = temp.row;

    sortMatrix();

    printf("Transpose completed.\n");
    display();

#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>

struct Node
{
    int data;
    struct Node *next;
};

struct Node *top = NULL;

void push(int value)
{
    struct Node *newNode = (struct Node *)malloc(sizeof(struct Node));

    if (newNode == NULL)
    {
        printf("Memory allocation failed!\n");
        exit(1);
    }

    newNode->data = value;
    newNode->next = top;
    top = newNode;
}

int pop()
{
    struct Node *temp;
    int value;

    if (top == NULL)
    {
        printf("Invalid postfix expression!\n");
        exit(1);
    }

    temp = top;
    value = temp->data;
    top = top->next;
    free(temp);

    return value;
}

int evaluate(char exp[])
{
    int i, a, b, result;

    for (i = 0; exp[i] != '\0'; i++)
    {
        if (isdigit((unsigned char)exp[i]))
        {
            push(exp[i] - '0');
        }
        else
        {
            b = pop();
            a = pop();

            switch (exp[i])
            {
                case '+':
                    result = a + b;
                    break;

                case '-':
                    result = a - b;
                    break;

                case '*':
                    result = a * b;
                    break;

                case '/':
                    if (b == 0)
                    {
                        printf("Division by zero!\n");
                        exit(1);
                    }
                    result = a / b;
                    break;

                default:
                    printf("Invalid operator!\n");
                    exit(1);
            }

            push(result);
        }
    }

    result = pop();

    if (top != NULL)
    {
        printf("Invalid postfix expression!\n");
        exit(1);
    }

    return result;
}

int values[100];
char operators[100];
int vtop = -1, otop = -1;

void pushValue(int x)
{
    values[++vtop] = x;
}

int popValue()
{
    return values[vtop--];
}

void pushOperator(char x)
{
    operators[++otop] = x;
}

char popOperator()
{
    return operators[otop--];
}

int precedence(char op)
{
    if (op == '+' || op == '-')
        return 1;

    if (op == '*' || op == '/' || op == '%')
        return 2;

    return 0;
}

void calculate()
{
    int a, b, result;
    char op;

    op = popOperator();
    b = popValue();
    a = popValue();

    switch (op)
    {
        case '+':
            result = a + b;
            break;

        case '-':
            result = a - b;
            break;

        case '*':
            result = a * b;
            break;

        case '/':
            if (b == 0)
            {
                printf("Division by zero!\n");

            break;

                exit(1);

            break;

            printf("Invalid operator!\n");
            exit(1);

}

code:#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *left, *right;
};

struct Node* createNode(int value)
{
    struct Node *newNode;

    newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = value;
    newNode->left = NULL;
    newNode->right = NULL;

    return newNode;
}

struct Node* insert(struct Node *root, int value)
{
    if (root == NULL)
        return createNode(value);

    if (value < root->data)
        root->left = insert(root->left, value);
    else if (value > root->data)
        root->right = insert(root->right, value);

    return root;
}

int height(struct Node *root)
{
    int left, right;

    if (root == NULL)
        return 0;

    left = height(root->left);
    right = height(root->right);

    if (left > right)
        return left + 1;
    else
        return right + 1;
}

int findMin(struct Node *root)
{
    if (root == NULL)
        return -1;

    while (root->left != NULL)
        root = root->left;

    return root->data;
}

struct Node* search(struct Node *root, int key)
{
    if (root == NULL || root->data == key)
        return root;

    if (key < root->data)
        return search(root->left, key);
    else
        return search(root->right, key);
}

void swapTree(struct Node *root)
{
    struct Node *temp;

    if (root == NULL)
        return;

    temp = root->left;
    root->left = root->right;
    root->right = temp;

    swapTree(root->left);
    swapTree(root->right);
}

void inorder(struct Node *root)
{
    if (root != NULL)
    {
        inorder(root->left);
        printf("%d ", root->data);
        inorder(root->right);
    }
}
